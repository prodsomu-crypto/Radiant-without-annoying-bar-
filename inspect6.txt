    34	// Where on an item (note or clip) spanning [x0, x1) pixels the mouse is.
    35	enum class Zone
    36	{
    37	    outside,
    38	    body,
    39	    leftEdge,
    40	    rightEdge
    41	};
    42	
    43	struct EdgeRules
    44	{
    45	    double grab = 6.0;             // widest edge zone, in pixels
    46	    double minimumHit = 4.0;       // very short items are still this wide to the mouse
    47	    double minimumForLeft = 16.0;  // narrower items only offer the right edge (the body must stay grabbable)
    48	    bool allowLeft = true;         // the left edge resizes / trims
    49	};
    50	
    51	// The edge zone shrinks with the item (a quarter of its width, at most `grab`), so a short note is
    52	// still mostly "body" and can be moved; the right edge always wins over the left one.
    53	inline Zone zoneAt (double x, double x0, double x1, const EdgeRules& rules = {})
    54	{
    55	    const auto right = std::max (x1, x0 + rules.minimumHit);
    56	
    57	    if (x < x0 || x >= right)
    58	        return Zone::outside;
    59	
    60	    const auto width = right - x0;
    61	    const auto edge = std::clamp (width / 4.0, 2.0, rules.grab);
    62	
    63	    if (x >= right - edge)
    64	        return Zone::rightEdge;
    65	    if (rules.allowLeft && width >= rules.minimumForLeft && x < x0 + edge)
    66	        return Zone::leftEdge;
    67	    return Zone::body;
    68	}

   129	// The action of a left press with `tool` on `target`. `playlist` selects the Playlist's extras
   130	// (slip, fades, inline automation, clip menu). Mirrors the editors' mouseDown.
   131	inline Action leftAction (EditTool tool, Target target, Mods mods, bool playlist)
   132	{
   133	    const auto onItem = target != Target::empty;
   134	
   135	    if (target == Target::clipMenu)
   136	        return Action::openMenu;
   137	
   138	    // every tool: Ctrl+click toggles an item in the selection; Ctrl+drag on empty space selects
   139	    if (mods.ctrl)
   140	        return onItem ? Action::toggleSelect : Action::boxSelect;
   141	
   142	    const auto painter = tool == EditTool::brush || tool == EditTool::drumPaint;
   143	
   144	    if (tool == EditTool::lasso || (! onItem && mods.shift && ! painter))
   145	        return Action::boxSelect;
   146	
   147	    switch (tool)
   148	    {
   149	        case EditTool::eraser: return Action::erase;
   150	        case EditTool::mute: return Action::toggleMute;
   151	        case EditTool::knife: return Action::slice;
   152	        case EditTool::zoom: return Action::zoomRange;
   153	        case EditTool::listen: return Action::listen;
   154	        case EditTool::slip: return playlist && onItem ? Action::slipContent : Action::none;
   155	        case EditTool::pencil:
   156	        case EditTool::brush:
   157	        case EditTool::drumPaint:
   158	        case EditTool::lasso: break;
   159	    }
   160	
   161	    if (playlist)
   162	    {
   163	        if (target == Target::autoPoint && ! mods.shift)
   164	            return Action::movePoint;
   165	        if (target == Target::autoTension && ! mods.shift)
   166	            return Action::bendSegment;
   167	        if (target == Target::autoCurve && ! mods.shift)
   168	            return Action::addPoint;
   169	        if (target == Target::fadeHandle)
   170	            return Action::fade;
   171	        if (target == Target::gainHandle)
   172	            return Action::clipGain;
   173	        if (mods.alt && mods.shift && target == Target::item)
   174	            return Action::slipContent;
   175	    }
   176	
   177	    switch (target)
   178	    {
   179	        case Target::item:
   180	        case Target::autoPoint:
   181	        case Target::autoCurve:
   182	        case Target::autoTension:
   183	        case Target::fadeHandle:
   184	        case Target::gainHandle: return mods.shift ? Action::copyMove : Action::move;
   185	        case Target::rightEdge: return Action::resizeRight;
   186	        case Target::leftEdge: return Action::resizeLeft;
   187	        case Target::empty: return painter ? Action::paint : Action::create;
   188	        case Target::clipMenu: break;
   189	    }
   190	
   191	    return Action::none;
   192	}

   248	// New scroll position (in ticks) after changing the zoom from `oldPixelsPerTick` to `newPixelsPerTick`,
   249	// keeping the tick under the mouse (`anchorPixels` from the left of the grid) where it is.
   250	inline double zoomAnchored (double scrollTicks, double oldPixelsPerTick, double newPixelsPerTick, double anchorPixels)
   251	{
   252	    const auto anchorTick = scrollTicks + anchorPixels / oldPixelsPerTick;
   253	    return std::max (0.0, anchorTick - anchorPixels / newPixelsPerTick);
   254	}

   256	// Zoom factor of one wheel event. Notched wheels step by 1.2 per notch whatever the driver's delta
   257	// size; smooth (trackpad / high-resolution) wheels zoom in proportion to the delta.
   258	inline double wheelZoomFactor (double deltaY, bool smooth)
   259	{
   260	    if (deltaY == 0.0)
   261	        return 1.0;
   262	    if (! smooth)
   263	        return deltaY > 0.0 ? 1.2 : 1.0 / 1.2;
   264	    return std::pow (1.2, std::clamp (deltaY, -1.0, 1.0) * 7.0);
   265	}
   266	
   267	// Scroll distance in pixels of one wheel event (notched: ~3 rows per notch; smooth: proportional).
   268	inline double wheelScrollPixels (double delta, bool smooth, double pixelsPerNotch)
   269	{
   270	    if (delta == 0.0)
   271	        return 0.0;
   272	    if (! smooth)
   273	        return (delta > 0.0 ? -1.0 : 1.0) * pixelsPerNotch;
   274	    return -delta * pixelsPerNotch * 7.0;
   275	}

   277	// --- trackpads (round 170e, Busy Works review #17: no quick zoom with a trackpad on macOS) ----------------------
   278	// Windows: precision touchpads (two-finger scrolling, and the pinch, which Windows sends as Ctrl+wheel) and
   279	// high-resolution wheels send WM_MOUSEWHEEL in pieces of a notch, and JUCE reports every piece as a notched event
   280	// (isSmooth false, one notch = 0.5 x 120 / 256 in its units): each piece then scrolled or zoomed a whole notch.
   281	// A delta under 3/4 of a notch is such a piece: it becomes a smooth delta of the same size (one notch = 1/7 in the
   282	// smooth formulas above), so a pinch zooms as far as the fingers move. Notched wheels are unchanged.
   283	constexpr double windowsWheelNotch = 60.0 / 256.0;
   284	
   285	struct WheelDelta
   286	{
   287	    double x = 0.0, y = 0.0;
   288	    bool smooth = false;
   289	};
   290	
   291	inline WheelDelta normaliseWheel (double deltaX, double deltaY, bool smooth, bool windows)
   292	{
   293	    const auto piece = [] (double d) { return d != 0.0 && std::abs (d) < 0.75 * windowsWheelNotch; };
   294	    if (smooth || ! windows || ! (piece (deltaX) || piece (deltaY)))
   295	        return { deltaX, deltaY, smooth };
   296	    return { deltaX / windowsWheelNotch / 7.0, deltaY / windowsWheelNotch / 7.0, true };
   297	}

   299	// The zoom factor of one trackpad pinch event (macOS magnifyWithEvent, Windows GID_ZOOM: JUCE's mouseMagnify `scale`,
   300	// 1 = no change), limited so that one event never jumps more than x2.
   301	inline double pinchZoomFactor (double scale)
   302	{
   303	    if (! (scale > 0.0) || ! std::isfinite (scale))
   304	        return 1.0;
   305	    return std::clamp (scale, 0.5, 2.0);
   306	}

   308	// --- middle-drag zooms (round 167t, issue #26; FL 2026 evidence in docs/analise/REA_rodada167t.md) ---------------
   309	// Key height range of the Piano roll. FL's is 18..116 px (16 + 2 px per zoom level 1..50); LYNX also goes
   310	// shorter, down to 6 px.
   311	constexpr int minKeyHeight = 6, maxKeyHeight = 116;
   312	
   313	// The vertical zoom drag: Ctrl+middle-drag on the Piano roll's notes (or the Playlist's clips) and middle-drag on
   314	// the Piano roll keyboard. Like FL's zoom knob it only follows the vertical travel: up = taller, one zoom level
   315	// (2 px) per 4 px of travel, per 16 px while Ctrl is held (fine). The travel is not clamped at the limits.
   316	struct HeightZoomDrag
   317	{
   318	    double travel = 0.0; // knob units (4 per pixel, 1 per pixel fine)
   319	    int lastY = 0;
   320	
   321	    void begin (int y) noexcept { travel = 0.0; lastY = y; }
   322	
   323	    void move (int y, bool fine) noexcept
   324	    {
   325	        travel += (double) (lastY - y) * (fine ? 1.0 : 4.0);
   326	        lastY = y;
   327	    }
   328	
   329	    // whole zoom levels gained since the press
   330	    int levels() const noexcept { return (int) std::lround (travel / 16.0); }
   331	
   332	    int height (int startHeight, int lo = minKeyHeight, int hi = maxKeyHeight) const noexcept
   333	    {
   334	        return std::clamp (startHeight + 2 * levels(), lo, hi);
   335	    }
   336	};

   338	// The horizontal zoom drag of FL's time scroll bar: middle-drag on the ruler, Alt+middle-drag in the grid.
   339	// Each move's vertical travel (at most 32 px a move) accumulates and the visible span becomes the span at the
   340	// press times 2^(-up / 100) (100 px up = twice the zoom) around the view's centre; the horizontal travel moves the
   341	// centre by half a view per 100 px (rightwards = later), except while Ctrl is held (zoom only).
   342	struct SpanZoomDrag
   343	{
   344	    double startSpan = 1.0, span = 1.0, centre = 0.0, up = 0.0;
   345	    int lastX = 0, lastY = 0;
   346	
   347	    void begin (double visibleSpan, double viewCentre, int x, int y) noexcept
   348	    {
   349	        startSpan = span = std::max (1.0e-9, visibleSpan);
   350	        centre = viewCentre;
   351	        up = 0.0;
   352	        lastX = x;
   353	        lastY = y;
   354	    }
   355	
   356	    void move (int x, int y, bool ctrl) noexcept
   357	    {
   358	        up += std::clamp ((double) (lastY - y), -32.0, 32.0);
   359	        span = startSpan * std::pow (2.0, -up * 0.01);
   360	        if (! ctrl)
   361	            centre += (double) (x - lastX) * span * 0.5 * 0.01;
   362	        centre = std::max (0.0, centre);
   363	        lastX = x;
   364	        lastY = y;
   365	    }
   366	};

   368	// A value that glides towards its target (smooth wheel scrolling). `step` advances it by one frame.
   369	struct Glide
   370	{
   371	    double current = 0.0, target = 0.0;
   372	
   373	    void jumpTo (double v) noexcept { current = target = v; }
   374	    bool isMoving() const noexcept { return current != target; }
   375	
   376	    // 60 Hz frames: covers ~45% of the remaining distance per frame, snaps the last half pixel
   377	    double step (double fraction = 0.45, double settle = 0.5) noexcept
   378	    {
   379	        current += (target - current) * fraction;
   380	        if (std::abs (target - current) < settle)
   381	            current = target;
   382	        return current;
   383	    }
   384	};

   386	// Value on the straight line from (x0, v0) to (x1, v1) at x (the event lane's right-drag line), 0..1.
   387	inline double lineValue (double x0, double v0, double x1, double v1, double x)
   388	{
   389	    if (x1 == x0)
   390	        return std::clamp (v1, 0.0, 1.0);
   391	    const auto t = (x - x0) / (x1 - x0);
   392	    return std::clamp (v0 + (v1 - v0) * t, 0.0, 1.0);
   393	}

   395	// FL's Piano roll keyboard: where a key is clicked sets the note's velocity, soft at the
   396	// key's base and full at its tip (`along` pixels from the base of a key `length` pixels long), 1/127 at least.
   397	inline float keyClickVelocity (double along, double length)
   398	{

   400	        return 0.8f;
   401	    return (float) std::clamp ((along + 1.0) / length, 1.0 / 127.0, 1.0);

   404	// Alt+wheel over a note: the property moves by 4/127 per notch (Ctrl+Alt: 1/127), clamped to 0..1.
   405	inline double wheelStepValue (double value, double deltaY, bool fine)
   406	{

   425	// --- overview strip (round 90) ------------------------------------------------------------------
   426	// The strip above the Piano Roll's and the Playlist's ruler shows the whole content in miniature with
   427	// the visible window outlined (FL Studio's preview scroll bar, interpreted). Positions are in content
   428	// units (ticks); `width` is the strip's width in pixels.
   429	enum class OverviewGrab { none, move, left, right };
   430	
   431	struct OverviewWindow
   432	{
   433	    double start = 0.0, length = 1.0;
   434	};
   435	
   436	// What a press at `x` grabs: an edge of the window (zooms), the window (scrolls) or nothing (jumps).
   437	inline OverviewGrab overviewGrabAt (double x, double width, double total, OverviewWindow w, double edgePixels = 5.0)
   438	{
   439	    if (total <= 0.0 || width <= 0.0)
   440	        return OverviewGrab::none;
   441	
   442	    const auto x0 = w.start / total * width, x1 = (w.start + w.length) / total * width;
   443	    // a narrow window keeps a middle zone to move it
   444	    const auto edge = std::min (edgePixels, std::max (1.0, (x1 - x0) / 3.0));
   445	
   446	    if (std::abs (x - x0) <= edge)
   447	        return OverviewGrab::left;
   448	    if (std::abs (x - x1) <= edge)
   449	        return OverviewGrab::right;
   450	    if (x > x0 && x < x1)
   451	        return OverviewGrab::move;
   452	    return OverviewGrab::none;
   453	}
   454	
   455	// The window after dragging `grab` by `dx` pixels from `from`; `minLength` keeps the zoom in range.
   456	// A press outside the window (none) centres it on the press first, then drags like `move`.
   457	inline OverviewWindow overviewDrag (OverviewGrab grab, OverviewWindow from, double dx, double width, double total,
   458	                                    double minLength)
   459	{
   460	    if (width <= 0.0 || total <= 0.0)
   461	        return from;
   462	
   463	    const auto d = dx / width * total;
   464	    auto w = from;
   465	
   466	    switch (grab)
   467	    {
   468	        case OverviewGrab::none:
   469	        case OverviewGrab::move:
   470	            w.start = std::clamp (from.start + d, 0.0, std::max (0.0, total - from.length));
   471	            break;
   472	        case OverviewGrab::left:
   473	        {
   474	            const auto end = from.start + from.length;
   475	            w.start = std::clamp (from.start + d, 0.0, std::max (0.0, end - minLength));
   476	            w.length = end - w.start;
   477	            break;
   478	        }
   479	        case OverviewGrab::right:
   480	            w.length = std::clamp (from.length + d, minLength, std::max (minLength, total - from.start));
   481	            break;
   482	    }
   483	
   484	    return w;
   485	}

   487	// The window centred on content position `at` (a click outside the window jumps there).
   488	inline OverviewWindow overviewCentredOn (double at, OverviewWindow w, double total)
   489	{
   490	    w.start = std::clamp (at - w.length / 2.0, 0.0, std::max (0.0, total - w.length));
   491	    return w;
   492	}
